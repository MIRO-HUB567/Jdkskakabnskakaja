			or "DISABLED"
	)

		addDetail("Logs", "No requests inspected yet.")
	else
		for i = #SecurityLogs, 1, -1 do
			local entry = SecurityLogs[i]

			local text = "[" .. entry.Action .. "]\n"
				.. "Source: " .. entry.Source .. "\n"
				.. "Method: " .. entry.Method .. "\n"

				text = text .. "\nMatch: " .. entry.Match
			end

			addDetail("#" .. tostring(entry.Id), text)
		end
	end

	Right.CanvasSize = UDim2.new(
		0,
		0,
		0,
		RightLayout.AbsoluteContentSize.Y + 15
	)
end

local function addSecurityLog(action, source, method, url, matched)
	SecurityCount += 1

	local entry = {
		Id = SecurityCount,
		Action = action,
		Source = source,
		Method = method,
		Url = url,
		Match = matched
	}

	table.insert(SecurityLogs, entry)

	if not SecurityViewing then
		SecurityUnread += 1
		updateUnread()
	end

	if REQUESTS_LOG then
		if action == "BLOCKED" then
			warn("[Anti IP Log] BLOCKED | " .. source .. " | " .. method .. " | " .. url)
		else
			print("[Anti IP Log] ALLOWED | " .. source .. " | " .. method .. " | " .. url)
		end
	end

	if SecurityViewing then
		task.defer(showSecurityLogs)
	end
end

local function normalize(value)
	local result = tostring(value or ""):lower()

	for _ = 1, 3 do
		local decoded = result:gsub("%%(%x%x)", function(hex)
			return string.char(tonumber(hex, 16))
		end)

		if decoded == result then
			break
		end

		result = decoded
	end

	result = result:gsub("\\/", "/")
	result = result:gsub("%s+", "")

	return result
end

local function containsPublicIPv4(value)
	local text = tostring(value or "")

	for a, b, c, d in text:gmatch("(%d+)%.(%d+)%.(%d+)%.(%d+)") do
		a, b, c, d = tonumber(a), tonumber(b), tonumber(c), tonumber(d)

		if a and b and c and d
			and a >= 1 and a <= 223
			and b >= 0 and b <= 255
			and c >= 0 and c <= 255
			and d >= 0 and d <= 255
			and a ~= 10
			and a ~= 127
			and not (a == 192 and b == 168)
			and not (a == 172 and b >= 16 and b <= 31)
			and not (a == 169 and b == 254) then
			return true, a .. "." .. b .. "." .. c .. "." .. d
		end
	end

	return false, nil
end

local function isSuspicious(url)
	local lower = normalize(url)

	for _, pattern in ipairs(suspiciousPatterns) do
		if string.find(lower, normalize(pattern), 1, true) then
			return true, pattern
		end
	end

	local pathPatterns = {
		"/myip", "/my-ip", "/publicip", "/public-ip",
		"/clientip", "/client-ip", "/remoteip", "/remote-ip",
		"/checkip", "/check-ip", "/getip", "/get-ip",
		"/whoami", "/webhook", "/webhooks"
	}

	for _, pattern in ipairs(pathPatterns) do
		if string.find(lower, pattern, 1, true) then
			return true, pattern
		end
	end

	if string.find(lower, "discord", 1, true)
		and string.find(lower, "webhook", 1, true) then
		return true, "discord webhook"
	end

	return false, nil
end

local function inspectResponse(response)
	if not INSPECT_RESPONSES or type(response) ~= "table" then
		return false, nil
	end

	local body = response.Body or response.body or ""
	body = tostring(body or "")

	local lower = normalize(body)
	local responsePatterns = {
		"your ip",
		"your_ip",
		"your-ip",
		"client ip",
		"client_ip",
		"client-ip",
		"remote ip",
		"remote_ip",
		"remote-ip",
		"public ip",
		"public_ip",
		"public-ip",
		"ip address",
		"ip_address",
		"ip-address"
	}

	for _, pattern in ipairs(responsePatterns) do
		if string.find(lower, pattern, 1, true) then
			local hasIP, ip = containsPublicIPv4(body)
			if hasIP then
				return true, "Response exposes public IP: " .. ip
			end
			return true, pattern
		end
	end

	return false, nil
end

local function inspectRequest(options, source)
	local url
	local method = "GET"

	if typeof(options) == "string" then
		url = options
	elseif typeof(options) == "table" then
		url = options.Url or options.URL or options.url
		method = options.Method or options.method or "GET"
	end

	url = tostring(url or "UNKNOWN")
	method = tostring(method)

	local suspicious, matched = isSuspicious(url)

	if suspicious and ANTI_IP_LOG then
		addSecurityLog("BLOCKED", source, method, url, matched)
		return true
	end

	addSecurityLog(
		suspicious and "ALLOWED" or "ALLOWED",
		source,
		method,
		url,
		suspicious and matched or nil
	)

	return false
end

local function spy(params)
	if inspectRequest(params, "request") then
		return {
			Success = false,
			StatusCode = 403,
			StatusMessage = "Blocked by Anti IP Log",
			Body = ""
		}
	end

	local success, result = pcall(function()
		return OriginalRequest(params)
	end)

	if success and INSPECT_RESPONSES and type(result) == "table" then
		local responseMatch, responseReason = inspectResponse(result)

		if responseMatch then
			local responseUrl = typeof(params) == "string"
				and params
				or typeof(params) == "table"
					and (params.Url or params.URL or params.url or "")
					or ""

			local responseMethod = typeof(params) == "table"
				and (params.Method or params.method or "GET")
				or "GET"

			addSecurityLog(
				"RESPONSE MATCH",
				"request",
				tostring(responseMethod),
				tostring(responseUrl),
				responseReason
			)
		end
	end

	LogCount += 1

	local response = nil

	if success and type(result) == "table" then
		response = result
	end

	local log = {
		Id = LogCount,

		Method = typeof(params) == "table"
			and (params.Method or params.method or "GET")
			or "GET",

		Url = typeof(params) == "string"
			and params
			or typeof(params) == "table"
				and (params.Url or params.URL or params.url or "")
				or "",

		Headers = typeof(params) == "table"
			and (params.Headers or {})
			or {},

		Body = typeof(params) == "table"
			and params.Body
			or nil,

		Response = response,
		StatusCode = response and response.StatusCode or "Error"
	}

	table.insert(Logs, log)
	addRequest(log)

	if success then
		return result
	end

	error(result, 0)
end

	end)
end

local startPosition

Top.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPosition = Main.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then
		return
	end

	if input.UserInputType ~= Enum.UserInputType.MouseMovement
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	local delta = input.Position - dragStart

	Main.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end)

Close.MouseButton1Click:Connect(function()
	Env.request = OriginalRequest
	Gui:Destroy()
end)

LeftLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
	Left.CanvasSize = UDim2.new(
		0,
		0,
		0,
		LeftLayout.AbsoluteContentSize.Y + 10
	)
end)

RightLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
	Right.CanvasSize = UDim2.new(
		0,
		0,
		0,
		RightLayout.AbsoluteContentSize.Y + 15
	)
end)

print("SS loaded")
